 boolean valid = switch (current) {
            case CREATED ->
                    next == DeliveryStatus.PREPARING
                            || next == DeliveryStatus.CANCELLED;

            case PREPARING ->
                    next == DeliveryStatus.SHIPPED
                            || next == DeliveryStatus.CANCELLED;

            case SHIPPED ->
                    next == DeliveryStatus.DELIVERED;

            case DELIVERED, CANCELLED -> false;
        };
